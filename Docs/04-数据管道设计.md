# 04 - 数据管道设计

## 完整抓取流程

```
关键词列表（task_queue）
        │
        ▼
┌─────────────────┐
│  1. 关键词搜索   │  search_keyword(keyword)
│  uiautomator2   │
└────────┬────────┘
         │
         ▼
┌─────────────────┐
│  2. 列表滚动     │  scroll_feed(count=N)
│  采集搜索结果    │  ← mitmproxy 自动拦截 API 响应
└────────┬────────┘
         │
         ▼
┌─────────────────┐
│  3. 进入帖子详情 │  tap(note_card)
│  采集详情数据    │  ← mitmproxy 拦截详情 API
└────────┬────────┘
         │
         ▼
┌─────────────────┐
│  4. 翻评论       │  open_comments() + scroll_comments()
│  采集评论数据    │  ← mitmproxy 拦截评论 API
└────────┬────────┘
         │
         ▼
┌─────────────────┐
│  5. 返回列表     │  tap_back() × 2
│  处理下一条      │
└────────┬────────┘
         │
    （全部帖子处理完）
         ▼
┌─────────────────┐
│  6. 任务完成     │  更新 crawl_tasks 状态
│  取下一个关键词   │
└─────────────────┘
```

## 任务队列设计

### src/orchestrator.py

```python
"""
任务调度主控 - 协调 uiautomator2 和 mitmproxy 完成数据采集
"""
import random
import time
import logging
from dataclasses import dataclass, field
from typing import Optional
from controller.device import connect_device, launch_app
from controller.actions import XHSActions
from controller.state import detect_page, PageState, navigate_to_home
from storage.db import Database

logger = logging.getLogger(__name__)


@dataclass
class CrawlTask:
    keyword: str
    target_notes: int = 20
    collect_comments: bool = True
    task_id: Optional[int] = None


class Orchestrator:
    def __init__(self, keywords: list[str]):
        self.db = Database()
        self.device = connect_device()
        self.actions = XHSActions(self.device)
        self.task_queue = [CrawlTask(kw) for kw in keywords]
        self.daily_count = 0

    def run(self, max_daily: int = 500):
        """主运行循环"""
        launch_app(self.device, fresh_start=True)
        time.sleep(2)

        for task in self.task_queue:
            if self.daily_count >= max_daily:
                logger.info(f"已达每日上限 {max_daily}，停止")
                break

            try:
                self._execute_task(task)
            except Exception as e:
                logger.error(f"任务异常 [{task.keyword}]: {e}")
                self._recover_to_home()

            # 关键词间随机等待
            delay = random.uniform(30, 90)
            logger.info(f"等待 {delay:.0f}s 后处理下一个关键词...")
            time.sleep(delay)

    def _execute_task(self, task: CrawlTask):
        """执行单个关键词任务"""
        logger.info(f"开始抓取关键词: {task.keyword}")

        # 创建任务记录
        task.task_id = self.db.create_task(task.keyword)

        # 搜索
        navigate_to_home(self.device)
        self.actions.search_keyword(task.keyword)
        time.sleep(2)

        # 滚动采集搜索结果（mitmproxy 在后台自动拦截）
        scroll_count = task.target_notes // 4  # 每次滑动约加载 4 条
        self.actions.scroll_feed(count=scroll_count)

        # 获取当前列表中的笔记（从数据库读取已拦截的数据）
        note_ids = self.db.get_uncrawled_notes(task.keyword, limit=task.target_notes)

        # 逐一进入详情页
        for i, note_id in enumerate(note_ids):
            if self.daily_count >= 500:
                break

            try:
                self._crawl_note(note_id, task.collect_comments)
                self.daily_count += 1
                logger.info(f"[{i+1}/{len(note_ids)}] 完成: {note_id}")
            except Exception as e:
                logger.warning(f"笔记 {note_id} 抓取失败: {e}")

        # 更新任务状态
        self.db.finish_task(task.task_id, total_notes=len(note_ids))

    def _crawl_note(self, note_id: str, collect_comments: bool):
        """抓取单个笔记的详情和评论"""
        # 通过 uiautomator2 找到对应笔记卡片并点击
        # 注：也可以通过 ADB intent 直接打开笔记（更快但特征明显）
        note_card = self.device(resourceId=f"note_{note_id}")
        if not note_card.exists(timeout=2):
            return

        self.actions.tap_element(note_card)
        self.actions.wait_for_page_load()

        # 在详情页停留随机时间（模拟阅读）
        time.sleep(random.uniform(3.0, 8.0))

        if collect_comments:
            if self.actions.open_comments():
                # 滚动评论
                scroll_count = random.randint(2, 5)
                self.actions.scroll_comments(scroll_count)

        self.actions.tap_back()
        if collect_comments:
            self.actions.tap_back()  # 关闭评论后还需返回列表

    def _recover_to_home(self):
        """异常恢复：回到首页"""
        try:
            navigate_to_home(self.device)
        except Exception:
            launch_app(self.device, fresh_start=True)
```

## 断点续采方案

### 设计原则
- **任务粒度**：以关键词为单位，完成一个关键词后更新状态
- **笔记去重**：note_id 作为主键，已采集的笔记不重复处理
- **崩溃恢复**：程序启动时检查 `crawl_tasks` 表中 `status=running` 的任务，从断点继续

### src/storage/db.py（断点相关）

```python
import sqlite3
from datetime import datetime
from pathlib import Path


class Database:
    def __init__(self, path: str = "data/xiaohongshu.db"):
        Path(path).parent.mkdir(parents=True, exist_ok=True)
        self.conn = sqlite3.connect(path, check_same_thread=False)
        self.conn.row_factory = sqlite3.Row
        self._init_tables()

    def _init_tables(self):
        self.conn.executescript("""
            CREATE TABLE IF NOT EXISTS notes (
                note_id TEXT PRIMARY KEY,
                title TEXT, desc TEXT,
                author_id TEXT, author_name TEXT,
                liked_count INTEGER DEFAULT 0,
                collected_count INTEGER DEFAULT 0,
                comment_count INTEGER DEFAULT 0,
                cover_url TEXT, note_type TEXT, topics TEXT,
                crawled_at TEXT,
                updated_at TEXT DEFAULT (datetime('now'))
            );
            CREATE TABLE IF NOT EXISTS comments (
                comment_id TEXT PRIMARY KEY,
                note_id TEXT, content TEXT,
                author_id TEXT, author_name TEXT,
                liked_count INTEGER DEFAULT 0,
                create_time INTEGER,
                parent_comment_id TEXT,
                crawled_at TEXT DEFAULT (datetime('now'))
            );
            CREATE TABLE IF NOT EXISTS crawl_tasks (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                keyword TEXT, status TEXT DEFAULT 'pending',
                total_notes INTEGER DEFAULT 0,
                created_at TEXT DEFAULT (datetime('now')),
                finished_at TEXT
            );
            CREATE INDEX IF NOT EXISTS idx_notes_author ON notes(author_id);
            CREATE INDEX IF NOT EXISTS idx_comments_note ON comments(note_id);
        """)
        self.conn.commit()

    def create_task(self, keyword: str) -> int:
        cur = self.conn.execute(
            "INSERT INTO crawl_tasks (keyword, status) VALUES (?, 'running')",
            (keyword,)
        )
        self.conn.commit()
        return cur.lastrowid

    def finish_task(self, task_id: int, total_notes: int):
        self.conn.execute(
            "UPDATE crawl_tasks SET status='done', total_notes=?, finished_at=? WHERE id=?",
            (total_notes, datetime.now().isoformat(), task_id)
        )
        self.conn.commit()

    def get_pending_tasks(self) -> list[str]:
        """获取未完成的关键词任务（用于断点续采）"""
        # 将中断的 running 任务重置为 pending
        self.conn.execute(
            "UPDATE crawl_tasks SET status='pending' WHERE status='running'"
        )
        cur = self.conn.execute(
            "SELECT keyword FROM crawl_tasks WHERE status='pending' ORDER BY id"
        )
        return [row[0] for row in cur.fetchall()]

    def get_uncrawled_notes(self, keyword: str, limit: int = 20) -> list[str]:
        """获取该关键词下还未抓取评论的笔记 ID"""
        # 从 mitmproxy 拦截的数据中找到笔记，筛选未处理的
        cur = self.conn.execute("""
            SELECT note_id FROM notes
            WHERE note_id NOT IN (SELECT DISTINCT note_id FROM comments)
            LIMIT ?
        """, (limit,))
        return [row[0] for row in cur.fetchall()]

    def save(self, items: list):
        """保存解析后的数据"""
        for item in items:
            if hasattr(item, 'note_id'):
                self._upsert_note(item)
            elif hasattr(item, 'comment_id'):
                self._upsert_comment(item)
        self.conn.commit()

    def _upsert_note(self, note):
        import json
        self.conn.execute("""
            INSERT OR REPLACE INTO notes
            (note_id, title, desc, author_id, author_name,
             liked_count, collected_count, comment_count,
             cover_url, note_type, topics, crawled_at)
            VALUES (?,?,?,?,?,?,?,?,?,?,?,?)
        """, (note.note_id, note.title, note.desc,
              note.author_id, note.author_name,
              note.liked_count, note.collected_count, note.comment_count,
              note.cover_url, note.note_type,
              json.dumps(note.topics, ensure_ascii=False),
              note.crawled_at))

    def _upsert_comment(self, comment):
        self.conn.execute("""
            INSERT OR REPLACE INTO comments
            (comment_id, note_id, content, author_id, author_name,
             liked_count, create_time, parent_comment_id)
            VALUES (?,?,?,?,?,?,?,?)
        """, (comment.comment_id, comment.note_id, comment.content,
              comment.author_id, comment.author_name,
              comment.liked_count, comment.create_time,
              comment.parent_comment_id))
```

## 数据去重逻辑

| 层级 | 去重方式 | 说明 |
|------|---------|------|
| 笔记 | `note_id` PRIMARY KEY + `INSERT OR REPLACE` | 同一笔记多次出现时更新数据而非重复插入 |
| 评论 | `comment_id` PRIMARY KEY + `INSERT OR REPLACE` | 同上 |
| 搜索任务 | 任务前查询 `crawl_tasks` 表 | 已完成的关键词不重复执行 |
| 原始 JSONL | 仅追加写入，不去重 | 保留完整原始记录便于调试 |

## 数据量估算

| 场景 | 估算 |
|------|------|
| 每次搜索滑动 | ~5 条/次滑动 |
| 每个关键词 | 20~25 条笔记 |
| 每条笔记评论 | 10~30 条 |
| 10 个关键词/天 | ~200~250 条笔记，~3000~5000 条评论 |
| SQLite 大小估算 | ~50MB/万条笔记 |
